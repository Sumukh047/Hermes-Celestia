async def execute_agent_turn(agent_id: str, prompt: str, session_id: str, max_turns: int = 8):
    """
    Hermes Multi-Turn Cognitive Execution Loop.
    Executes iterative ReAct cycles against local LM Studio instance.
    """
    agent = await get_agent_by_id(agent_id)
    history = await load_session_history(session_id)
    tools = await get_registered_tools_for_agent(agent_id)

    # 1. Compile System Prompt with Persona & Tool Schemas
    system_prompt = build_grounded_system_prompt(agent, tools)
    messages = [{"role": "system", "content": system_prompt}] + history + [{"role": "user", "content": prompt}]

    turn = 0
    await event_hub.broadcast(session_id, {"event": "agent_start", "agent_id": agent_id, "prompt": prompt})

    while turn < max_turns:
        turn += 1
        # 2. Call Local LM Studio Inference Engine (Zero egress)
        response = await lm_client.chat.completions.create(
            model=agent.model_name or "local-model",
            messages=messages,
            temperature=agent.temperature or 0.3,
            tools=[tool.openai_schema for tool in tools] if tools else None,
            tool_choice="auto" if tools else "none"
        )

        choice = response.choices[0]
        # Emit reasoning / thought telemetry
        if choice.message.content:
            await event_hub.broadcast(session_id, {
                "event": "agent_thought",
                "turn": turn,
                "content": choice.message.content
            })

        # 3. Check for Tool Invocation Requests
        if choice.message.tool_calls:
            for tool_call in choice.message.tool_calls:
                tool_name = tool_call.function.name
                args = json.loads(tool_call.function.arguments)

                # 4. Check Human-in-the-Loop (HITL) Policy
                if is_destructive_action(tool_name, args):
                    approval = await request_hitl_approval(session_id, tool_name, args)
                    if not approval.approved:
                        messages.append({"role": "tool", "name": tool_name, "content": "Action rejected by human operator."})
                        continue

                # 5. Execute Grounded Tool
                await event_hub.broadcast(session_id, {"event": "tool_start", "tool": tool_name, "args": args})
                result = await tool_registry.dispatch(tool_name, args)
                await event_hub.broadcast(session_id, {"event": "tool_finish", "tool": tool_name, "result": result})

                messages.append({"role": "tool", "name": tool_name, "content": json.dumps(result)})
        else:
            # Final Natural Language Answer Reached
            await event_hub.broadcast(session_id, {
                "event": "agent_complete",
                "final_answer": choice.message.content,
                "total_turns": turn
            })
            await persist_message(session_id, "assistant", choice.message.content)
            return choice.message.content

    return "Max cognitive turns reached without convergence."
def build_grounded_system_prompt(agent, tools) -> str:
    """
    Compiles anti-hallucination system prompt grounded in local memory and active tools.
    """
    prompt_parts = [
        f"You are {agent.name}, an autonomous agent operating within Hermes Agent Studio.",
        f"AGENT UUID: {agent.id}",
        f"ROLE & PERSONA: {agent.system_prompt or 'You are an intelligent, factual AI assistant.'}",
        "",
        "### STRICT GROUNDING & ANTI-HALLUCINATION RULES:",
        "1. Never invent email senders, subjects, dates, or body text. Rely ONLY on data returned by get_unread_emails or search_inbox.",
        "2. If an email body contains no actionable content, explicitly state: 'No actionable content found.'",
        "3. When invoking tools, supply valid JSON matching the exact schema properties.",
        "4. All thoughts must be concise, analytical, and focused on accomplishing the objective.",
        "",
        "### AVAILABLE TOOLS CATALOG:"
    ]
    for tool in tools:
        prompt_parts.append(f"- {tool.name}: {tool.description}")
        prompt_parts.append(f"  Parameters: {json.dumps(tool.parameters)}")

    return "\n".join(prompt_parts)
DESTRUCTIVE_TOOLS = {
    "execute_system_command",
    "send_email_reply",
    "delete_database_record",

def is_destructive_action(tool_name: str, args: dict) -> bool:
    """
    Determines whether a tool action requires human oversight.
    """
    if tool_name in DESTRUCTIVE_TOOLS:
        return True
    if tool_name == "sql_query" and any(k in args.get("query", "").lower() for k in ["drop", "truncate", "delete", "alter"]):
        return True
    return False

async def request_hitl_approval(session_id: str, tool_name: str, args: dict) -> ApprovalResult:
    token = str(uuid.uuid4())
    await event_hub.broadcast(session_id, {
        "event": "hitl_requested",
        "approval_token": token,
        "tool": tool_name,
        "arguments": args
    })
    # Suspends execution until user clicks 'Approve' in UI or via Discord modal
    return await approval_queue.wait_for_decision(token, timeout_seconds=120)
class EventHub:
    """
    In-memory asynchronous pub/sub broadcaster for real-time Agent UI streaming.
    """
    def __init__(self):
        self._subscribers: dict[str, set[asyncio.Queue]] = defaultdict(set)

    async def subscribe(self, session_id: str) -> asyncio.Queue:
        q = asyncio.Queue(maxsize=100)
        self._subscribers[session_id].add(q)
        return q

    async def unsubscribe(self, session_id: str, q: asyncio.Queue):
        if session_id in self._subscribers and q in self._subscribers[session_id]:
            self._subscribers[session_id].remove(q)

    async def broadcast(self, session_id: str, payload: dict):
        payload["timestamp"] = datetime.utcnow().isoformat() + "Z"
        if session_id in self._subscribers:
            for q in list(self._subscribers[session_id]):
                try:
                    q.put_nowait(payload)
                except asyncio.QueueFull:
                    pass

2. Universal RFC 822 MIME Parser & Anti-Hallucination Grounding

A common pitfall with LLM-powered email assistants is hallucination — when an LLM is given an empty or unparsed HTML email body, it invents generic subjects like "Exciting Project Opportunity" or "Meeting Next Week". Hermes solves this completely with our deterministic multipart MIME extraction engine.

📄 backend/main.py: extract_email_body_clean()
def extract_email_body_clean(msg: email.message.Message) -> str:
    """
    Recursively decodes multipart RFC 822 emails.
    Prioritizes text/plain over text/html to avoid injecting CSS/JS.
    """
    body = ""
    if msg.is_multipart():
        for part in msg.walk():
            ctype = part.get_content_type()
            cdisp = str(part.get("Content-Disposition", ""))
            if ctype == "text/plain" and "attachment" not in cdisp:
                payload = part.get_payload(decode=True)
                if payload:
                    charset = part.get_content_charset() or "utf-8"
                    body = payload.decode(charset, errors="replace")
                    break
        # Fallback to HTML parsing if text/plain absent
        if not body:
            for part in msg.walk():
                if part.get_content_type() == "text/html":
                    payload = part.get_payload(decode=True)
                    if payload:
                        charset = part.get_content_charset() or "utf-8"
                        raw_html = payload.decode(charset, errors="replace")
                        body = BeautifulSoup(raw_html, "html.parser").get_text()
                        break
    else:
        payload = msg.get_payload(decode=True)
        if payload:
            charset = msg.get_content_charset() or "utf-8"
            body = payload.decode(charset, errors="replace")

    # Clean whitespace and normalize lines
    lines = [line.strip() for line in body.splitlines() if line.strip()]
    return "\n".join(lines)[:4000]

1. Background Scheduler Engine & Natural Language Parsing

Rather than requiring end users to write cryptic cron syntax like */15 * * * *, Hermes implements an intelligent interval compiler that accepts both natural English phrases and standard cron expressions.

📄 backend/main.py: scheduler_worker() Loop & Interval Evaluator
async def scheduler_worker():
    """
    Continuous 15-second tick loop inspecting configured automations in SQLite.
    Executes grounded agent prompts and logs execution history.
    """
    while True:
        try:
            await asyncio.sleep(15)
            async with aiosqlite.connect(DB_PATH) as db:
                db.row_factory = aiosqlite.Row
                async with db.execute("SELECT * FROM automations WHERE status = 'active'") as cur:
                    jobs = await cur.fetchall()

            now = datetime.utcnow()
            for job in jobs:
                if should_execute_job(job["schedule_expr"], job["last_run"], now):
                    # Mark status as running
                    await update_job_status(job["id"], "running")
                    
                    # Execute agent turn with factual grounding
                    output = await execute_agent_turn(
                        agent_id=job["agent_id"],
                        prompt=job["prompt"],
                        session_id=f"auto_{job['id']}_{int(now.timestamp())}"
                    )
                    
                    # Persist last result and execution timestamp
                    await record_job_completion(job["id"], output, now)
        except Exception as e:
            logger.error(f"Scheduler daemon error: {e}", exc_info=True)

def should_execute_job(expr: str, last_run_str: str | None, now: datetime) -> bool:
    """
    Parses expressions like 'every minute', 'every 30 minutes', 'every 2 hours', 'daily at 09:00'.
    """
    if not last_run_str:
        return True
    last_run = datetime.fromisoformat(last_run_str)
    elapsed = (now - last_run).total_seconds()

    m = re.match(r"every\s+(\d+)?\s*(minute|hour|day)s?", expr.lower().strip())
    if m:
        val = int(m.group(1)) if m.group(1) else 1
        unit = m.group(2)
        multiplier = 60 if unit == "minute" else 3600 if unit == "hour" else 86400
        return elapsed >= (val * multiplier)

    # Standard Cron Expression evaluation fallback
    return croniter(expr, last_run).get_next(datetime) <= now

LM Studio's local server runs at http://127.0.0.1:1234/v1. When started via batch or when the frontend was served from a disparate port, browser CORS headers blocked the frontend from polling port 1234 directly. Hermes resolved this by routing all model detection requests through FastAPI's server-side /api/lmstudio/status proxy. The proxy issues an asynchronous HTTP request from Python (bypassing browser CORS) to 127.0.0.1:1234/v1/models. If LM Studio is loaded, it returns the active model name and sets the dashboard indicator to green.
Discord's API requires bots to explicitly declare Privileged Gateway Intents to read message bodies. Without intent mask MESSAGE_CONTENT (32768), Discord delivers empty strings for content in all MESSAGE_CREATE dispatch events. Hermes resolved this in backend/main.py by sending intent mask 37377 (which includes GUILDS | GUILD_MESSAGES | DIRECT_MESSAGES | MESSAGE_CONTENT) during the Opcode 2 IDENTIFY handshake, and establishing an Opcode 1 heartbeat loop every 41.25s to keep the bot green.
Naive LLM integrations pass raw unparsed email blobs or prompt the LLM without context, causing the model to invent plausible-sounding emails like "Exciting New Project Opportunity". Hermes implemented extract_email_body_clean(), which recursively traverses multipart RFC 822 MIME structures, selects pure text/plain or cleans text/html with BeautifulSoup, strips email signatures and tracking links, and injects the literal sender, date, and body text directly into the agent's prompt with strict rules forbidding invention.
Normally, React Vite dev runs on port 5173 while FastAPI runs on port 8790. Exposing this via ngrok requires either two paid tunnels or complicated reverse proxies. Hermes builds the Vite project into frontend/dist and mounts it as a static folder inside FastAPI using StaticFiles(directory="frontend/dist", html=True). As a result, both the UI, REST APIs, and WebSockets share port 8790, requiring only 1 single Ngrok tunnel.
Traditional queues require provisioning and paying for dedicated Redis or RabbitMQ instances, managing worker concurrency pools, and writing boilerplate retry and serialization logic. Render Workflows provides a code-first durable execution model (render.com/workflows). Decorated tasks (@app.task) automatically snapshot state, persist return values, and retry on failure with exponential backoff without requiring external broker infrastructure.
